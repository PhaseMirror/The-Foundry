Key functions
python
# k3_container.py — stdlib only

import hashlib, json, os, struct, argparse, datetime
from pathlib import Path

DOMAIN = b"PM-K2-ACCEPT-v1"

def sha256_bytes(b: bytes) -> str:
    return "sha256:" + hashlib.sha256(b).hexdigest()

def store_blob(vault: Path, b: bytes) -> str:
    cid = sha256_bytes(b)
    hexdigest = cid.split(":", 1)[1]
    p = vault / "blobs" / hexdigest
    if not p.exists():
        p.write_bytes(b)
    return cid

def accept_preimage(envelope_id: str, timestamp: int, doc_digest_hex: str) -> bytes:
    eid = envelope_id.encode("utf-8")
    digest = bytes.fromhex(doc_digest_hex)
    if len(digest) != 32:
        raise ValueError("doc_digest must be 32 bytes")
    return (
        DOMAIN
        + struct.pack("<I", len(eid))
        + eid
        + struct.pack("<Q", timestamp)
        + digest
    )
The export gate
The vault refuses to export a preimage unless all three hold:

def may_export_preimage(entry: dict) -> bool:
    return (
        entry["acceptPath"] == "049c"
        and entry["pinned"] is True
        and entry.get("pinAdr") is not None
    )
acceptPath == "049c" — the entry was added with an Accept path.

pinned is True — a frozen canonical artifact exists.

pinAdr — the ADR that froze it is recorded.

No seat required for the gate itself. The gate is local. The seat signs after the gate opens.

python
def cmd_export_preimage(vault, entry_id, envelope_id, timestamp=None):
    entry = load_entry(vault, entry_id)
    if not may_export_preimage(entry):
        raise SystemExit(
            "refused: bytes_pinned=false or acceptPath != 049c. "
            "Pin the canonical bytes first."
        )
    ts = timestamp or int(datetime.datetime.utcnow().timestamp())
    pre = accept_preimage(envelope_id, ts, entry["cid"].split(":", 1)[1])
    out = vault / "preimages" / f"{entry_id}.bin"
    out.write_bytes(pre)
    entry["exports"].append({
        "envelopeId": envelope_id,
        "timestamp": ts,
        "preimageDigest": sha256_bytes(pre),
        "exportedAt": datetime.datetime.utcnow().isoformat() + "Z"
    })
    save_entry(vault, entry)
    return out
The preimage is unsigned at export. Signing is a separate step by a separate door.

Recording signatures

def cmd_record_signature(vault, entry_id, role, pub_path, sig_path):
    entry = load_entry(vault, entry_id)
    pub = Path(pub_path).read_bytes()
    sig = Path(sig_path).read_bytes()
    # verify against the preimage bytes before recording
    pre_path = vault / "preimages" / f"{entry_id}.bin"
    if not pre_path.exists():
        raise SystemExit("refused: no exported preimage for this entry")
    if not verify_ed25519(pub, pre_path.read_bytes(), sig):
        raise SystemExit("refused: signature does not verify")
    receipts = load_receipts(vault, entry_id)
    if any(s["pubkey"] == pub.hex() for s in receipts["signatures"]):
        raise SystemExit("refused: this key already signed")
    receipts["signatures"].append({
        "role": role,
        "pubkey": pub.hex(),
        "sig": sig.hex(),
        "signedAt": datetime.datetime.utcnow().isoformat() + "Z"
    })
    # accept only when two DISTINCT pubkeys signed the SAME preimage digest
    distinct = {s["pubkey"] for s in receipts["signatures"]}
    receipts["accept"] = len(distinct) >= 2
    save_receipts(vault, entry_id, receipts)
    return receipts
Two distinct keys, same preimage digest. That is Accept. One key is a proposal.

Freeze: producing canonical bytes
python
def cmd_freeze(vault, entry_id, out_name, source_docx=None):
    entry = load_entry(vault, entry_id)
    src = vault / "blobs" / entry["cid"].split(":", 1)[1]
    raw = src.read_bytes()
    # canonical form: UTF-8, NFC, LF, no trailing whitespace
    import unicodedata
    text = raw.decode("utf-8")
    text = unicodedata.normalize("NFC", text)
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = "\n".join(line.rstrip() for line in text.split("\n"))
    canonical = text.encode("utf-8")
    cid = store_blob(vault, canonical)
    entry["canonicalCid"] = cid
    entry["canonicalName"] = out_name
    entry["provenance"] = {
        "sourceCid": entry["cid"],
        "sourceDocx": source_docx,
        "frozenAt": datetime.datetime.utcnow().isoformat() + "Z"
    }
    save_entry(vault, entry)
    return cid
pin is a separate command that records the ADR and flips pinned=true. Freeze produces bytes; pin asserts they are the canonical 049c text. Two distinct acts.
