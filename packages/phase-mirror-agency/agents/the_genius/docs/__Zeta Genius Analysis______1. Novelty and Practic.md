  def finite_diff_jacobian(transfer_fn, psi_baseline, primes, eps=EPSILON_JACOBIAN):
      """Estimate Jacobian of transfer_fn at prime frequencies via finite difference."""
      lipschitz_estimates = {}
      for p in primes:
          psi_perturbed = {k: v + (eps if k == p else 0) for k, v in psi_baseline.items()}
          delta_out = transfer_fn(psi_perturbed)[p] - transfer_fn(psi_baseline)[p]
          lipschitz_estimates[p] = abs(delta_out) / eps
      return lipschitz_estimates
