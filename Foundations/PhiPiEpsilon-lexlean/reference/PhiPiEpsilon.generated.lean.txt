public inductive Intent where
  | exploratory
  | declarative
  | reflective
  | ambiguous
  | coercive
  | forcedResolution
  | paradoxSpiral

public inductive PauseReason where
  | forcedResolution
  | paradoxSpiral

public inductive RefusalReason where
  | safeBoundary
  | riskAtOrAboveThreshold

public inductive Decision where
  | reflect
  | pause (_ : PauseReason)
  | refuse (_ : RefusalReason)

public inductive Threshold where
  | below
  | atOrAbove

public inductive DecisionKind where
  | normal
  | pause
  | refusal

public structure ExternalAction where
  id : Nat

public structure Authorization where
  actionId : Nat
  expiresAt : Nat
  revoked : Bool

public structure Context where
  intent : Intent
  pressure : Nat
  recursionDepth : Nat

public structure Assessment where
  intent : Intent
  riskScore : Nat

public structure SystemState where
  riskThreshold : Nat
  authorization : Option (Authorization)
  now : Nat

@[expose] public def Decision.kind (d : Decision) : DecisionKind := (match d with | Decision.reflect => DecisionKind.normal | Decision.pause _ => DecisionKind.pause | Decision.refuse _ => DecisionKind.refusal)

@[expose] public def isReflect (d : Decision) : Bool := (match d with | Decision.reflect => true | Decision.pause _ => false | Decision.refuse _ => false)

@[expose] public def isPause (d : Decision) : Bool := (match d with | Decision.reflect => false | Decision.pause _ => true | Decision.refuse _ => false)

@[expose] public def isRefusal (d : Decision) : Bool := (match d with | Decision.reflect => false | Decision.pause _ => false | Decision.refuse _ => true)

@[expose] public def isOrdinary (i : Intent) : Bool := (match i with | Intent.exploratory => true | Intent.declarative => true | Intent.reflective => true | Intent.ambiguous => true | Intent.coercive => false | Intent.forcedResolution => false | Intent.paradoxSpiral => false)

@[expose] public def isPauseIntent (i : Intent) : Bool := (match i with | Intent.exploratory => false | Intent.declarative => false | Intent.reflective => false | Intent.ambiguous => false | Intent.coercive => false | Intent.forcedResolution => true | Intent.paradoxSpiral => true)

@[expose] public def baseRisk (i : Intent) : Nat := (match i with | Intent.exploratory => 0 | Intent.declarative => 0 | Intent.reflective => 1 | Intent.ambiguous => 3 | Intent.coercive => 10 | Intent.forcedResolution => 10 | Intent.paradoxSpiral => 10)

@[expose] public def assess (ctx : Context) : Assessment := ({ intent := (ctx).intent, riskScore := ((baseRisk ((ctx).intent) + (ctx).pressure) + (ctx).recursionDepth) } : Assessment)

@[expose] public def thresholdOf (risk : Nat) (threshold : Nat) : Threshold := (match (Nat.blt (risk) (threshold)) with | Bool.true => Threshold.below | Bool.false => Threshold.atOrAbove)

@[expose] public def isBelow (t : Threshold) : Bool := (match t with | Threshold.below => true | Threshold.atOrAbove => false)

@[expose] public def routeWith (i : Intent) (t : Threshold) : Decision := (match i with | Intent.exploratory => (match t with | Threshold.below => Decision.reflect | Threshold.atOrAbove => Decision.refuse (RefusalReason.riskAtOrAboveThreshold)) | Intent.declarative => (match t with | Threshold.below => Decision.reflect | Threshold.atOrAbove => Decision.refuse (RefusalReason.riskAtOrAboveThreshold)) | Intent.reflective => (match t with | Threshold.below => Decision.reflect | Threshold.atOrAbove => Decision.refuse (RefusalReason.riskAtOrAboveThreshold)) | Intent.ambiguous => (match t with | Threshold.below => Decision.reflect | Threshold.atOrAbove => Decision.refuse (RefusalReason.riskAtOrAboveThreshold)) | Intent.coercive => Decision.refuse (RefusalReason.safeBoundary) | Intent.forcedResolution => Decision.pause (PauseReason.forcedResolution) | Intent.paradoxSpiral => Decision.pause (PauseReason.paradoxSpiral))

@[expose] public def route (st : SystemState) (a : Assessment) : Decision := routeWith ((a).intent) (thresholdOf ((a).riskScore) ((st).riskThreshold))

@[expose] public def decideFor (st : SystemState) (ctx : Context) : Decision := route (st) (assess (ctx))

@[expose] public def Authorization.isValidFor (auth : Authorization) (action : ExternalAction) (now : Nat) : Bool := (((!(auth).revoked) && (Nat.beq ((auth).actionId) ((action).id))) && (Nat.blt (now) ((auth).expiresAt)))

@[expose] public def authorizedFor (authorization : Option (Authorization)) (action : ExternalAction) (now : Nat) : Bool := (match authorization with | Option.none => false | Option.some auth => Authorization.isValidFor (auth) (action) (now))

@[expose] public def mayPerformExternalAction (st : SystemState) (d : Decision) (action : ExternalAction) : Bool := (match d with | Decision.reflect => authorizedFor ((st).authorization) (action) ((st).now) | Decision.pause _ => false | Decision.refuse _ => false)

  rfl
